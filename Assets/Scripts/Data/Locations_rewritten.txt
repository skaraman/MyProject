    // Data-only location configuration - no runtime logic, GameObject refs stored as paths only
    
    [Serializable] public partial struct EnemySpawnRule : IComparable<EnemySpawnRule> {
        public int MaxCount = 0;
        public float IntervalSeconds = 1f;
        
        public EnemySpawnRule(int maxEnemies, float spawnInterval) {
            this.MaxCount = maxEnemies; 
            this.IntervalSeconds = spawnInterval;
        }

        public int CompareTo(EnemySpawnRule other) => MaxCount.CompareTo(other.MaxCount);
    }
